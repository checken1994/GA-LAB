## D6.1 Scanner source (self-contained, copy-paste reproducible)
"""D6 fail-loudly scanner — silent `except: pass` detector (AST, no execution).

Contract scope (decides exit code):
  PASS_ONLY / EMPTY — handlers that swallow the exception itself with a
  pass-only or empty body and no log/raise/print. These are the D6 violation.

Informational (does not decide exit code, printed for transparency):
  FALLBACK_NO_LOG — handlers with real statements but no log/raise/print
  (e.g. `except Exception: value = "unknown"`). Out of the D6 `except: pass`
  contract; listed so the reader sees the exact coverage boundary.

Usage: py -3.12 scp_m1_ast_scan.py <file> [<file> ...]
Exit:  0 = zero contract violations, 1 = at least one contract violation.
"""
import ast
import sys

LOUD_CALLS = {"warning", "error", "critical", "exception", "info", "debug", "print"}
CONTRACT = "D6_SCOPE"
INFO = "OUT_OF_SCOPE"


def _is_observable(node: ast.AST) -> bool:
    for child in ast.walk(node):
        if isinstance(child, ast.Raise):
            return True
        if isinstance(child, ast.Call):
            fn = child.func
            name = None
            if isinstance(fn, ast.Attribute):
                name = fn.attr
            elif isinstance(fn, ast.Name):
                name = fn.id
            if name in LOUD_CALLS:
                return True
    return False


def scan(path: str) -> list[tuple[str, int, str, str]]:
    with open(path, "r", encoding="utf-8") as fh:
        src = fh.read()
    tree = ast.parse(src, filename=path)
    findings: list[tuple[str, int, str, str]] = []
    for node in ast.walk(tree):
        if not isinstance(node, ast.ExceptHandler):
            continue
        body = [s for s in node.body if not (isinstance(s, ast.Expr)
                and isinstance(s.value, ast.Constant) and isinstance(s.value.value, str))]
        exc = ast.unparse(node.type) if node.type is not None else "bare-except"
        if not body:
            findings.append((CONTRACT, node.lineno, exc, "empty body"))
            continue
        if all(isinstance(s, ast.Pass) for s in body):
            findings.append((CONTRACT, node.lineno, exc, "pass-only body"))
            continue
        if not any(_is_observable(s) for s in body):
            findings.append((INFO, node.lineno, exc, "fallback/action without log"))
    return findings


def main(argv: list[str]) -> int:
    paths = argv[1:]
    violations = 0
    infos = 0
    for path in paths:
        findings = scan(path)
        own_v = [f for f in findings if f[0] == CONTRACT]
        own_i = [f for f in findings if f[0] == INFO]
        violations += len(own_v)
        infos += len(own_i)
        for _, lineno, exc, why in own_v:
            print(f"VIOLATION  {path}:{lineno}  except {exc}  ({why})")
        for _, lineno, exc, why in own_i:
            print(f"INFO       {path}:{lineno}  except {exc}  ({why})")
        if not own_v:
            print(f"OK         {path}  (0 except:pass handlers, {len(own_i)} out-of-scope fallback handlers)")
    print()
    print(f"CONTRACT_VIOLATIONS_EXCEPT_PASS_WITHOUT_LOG={violations}")
    print(f"OUT_OF_SCOPE_FALLBACK_NO_LOG={infos}")
    print(f"SCANNED_FILES={len(paths)}")
    return 1 if violations else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
