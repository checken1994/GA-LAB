## D6.1 Probe source (self-contained, copy-paste reproducible)
"""D6 runtime probe (re-run at re-pin) — does the fail-loudly SCP_PORT branch emit safely?

In-process, using a capture handler + a formatter-identical stream handler on the
"scp.api" logger, so the exact *physical* log lines are observable:

  A. malformed value       -> WARNING [MACH1-FIX-8] emitted, fallback preserved
  B. log-injection attempt -> value carrying a newline + a fake "CRITICAL" line is
                              sanitised: the emitted record stays ONE physical line
                              and the payload tail never reaches the log
  C. clean environment     -> no [MACH1-FIX-8] record at all (no false positive)

Exit 0 only if every assertion holds; prints PASS/FAIL per case.
"""
import io
import logging
import os
import sys

records = []
buf = io.StringIO()
fmt = logging.Formatter("%(asctime)s | %(levelname)-7s | %(name)s | %(message)s")


class Capture(logging.Handler):
    def emit(self, record):
        records.append(record)


log = logging.getLogger("scp.api")
log.addHandler(Capture())
stream = logging.StreamHandler(buf)
stream.setFormatter(fmt)
log.addHandler(stream)
log.setLevel(logging.DEBUG)

from scp.api_server import _scp_service_identity  # noqa: E402  (import after handlers)

INJECTION = "8000\n2026-09-10 00:00:00 | CRITICAL | fake-log-line"
RAW_SENTINEL = "CRITICAL | fake-log-line"

failures = []


def probe(value):
    records.clear()
    buf.seek(0)
    buf.truncate(0)
    if value is None:
        os.environ.pop("SCP_PORT", None)
    else:
        os.environ["SCP_PORT"] = value
    identity = _scp_service_identity()
    warned = [r.getMessage() for r in records if "[MACH1-FIX-8]" in r.getMessage()]
    return identity["configured_port"], warned, buf.getvalue()


def case(label, value, expect_warning):
    port, warned, physical = probe(value)
    print("--- case %s: SCP_PORT=%r" % (label, value))
    print("    physical_lines=%d  warnings=%d  configured_port=%s"
          % (physical.count("\n"), len(warned), port))
    for line in physical.splitlines():
        print("    LINE %r" % line)
    if expect_warning:
        if not warned:
            failures.append("%s: no [MACH1-FIX-8] warning emitted" % label)
        if port != 8000:
            failures.append("%s: fallback broken, configured_port=%r" % (label, port))
        if physical.count("\n") != len(warned):
            failures.append("%s: %d physical lines for %d records (multi-line record)"
                            % (label, physical.count("\n"), len(warned)))
        if any(("\n" in w or "\r" in w) for w in warned):
            failures.append("%s: warning message contains a control character" % label)
    else:
        if warned:
            failures.append("%s: unexpected [MACH1-FIX-8] warning on a clean env" % label)
        if port != 8000:
            failures.append("%s: clean-env port=%r" % (label, port))
    return port, warned, physical


case("A malformed", "not-a-number", True)

port_b, warned_b, physical_b = case("B injection", INJECTION, True)
if any(RAW_SENTINEL in w for w in warned_b):
    failures.append("B: raw injected payload reached the log message")
if RAW_SENTINEL in physical_b:
    failures.append("B: raw injected payload reached the physical log output")

case("C clean env", None, False)

print()
print("ASSERTIONS: %d failure(s)" % len(failures))
for f in failures:
    print("FAIL: %s" % f)
print("PROBE_RESULT=%s" % ("FAIL" if failures else "PASS"))
for extra in ("SCP_PORT=%r" % INJECTION,):
    print("payload_used = %s" % extra)
sys.exit(1 if failures else 0)

## D6.3 What this proves / does not prove
PROVES (runtime, not static):
  - the fail-loudly branch really emits at WARNING level with the [MACH1-FIX-8] tag (case A);
  - the fallback behaviour is unchanged: configured_port stays 8000 when SCP_PORT is malformed (A,B)
    and when SCP_PORT is unset (C);
  - log hygiene (F7): a value carrying a newline + a fake "CRITICAL" line stays ONE physical log
    line, the newline is replaced by "?", the diagnostic preview is truncated to 24 chars, and the
    injected payload tail never reaches the message or the physical output (case B);
  - no false positive: a clean env emits no such warning (case C).
DOES NOT PROVE:
  - that other env-derived strings anywhere else in the codebase are sanitised (only this call site
    was changed);
  - that a clean boot in production never trips this branch (D2 covers the clean-boot path).
